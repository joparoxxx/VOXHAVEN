cmake_minimum_required(VERSION 3.22)
project(cubyte_llama LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/deps/godot-cpp" godot-cpp)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/deps/llama.cpp" llama)
add_library(cubyte_llama SHARED src/bridge.cpp)
target_link_libraries(cubyte_llama PRIVATE godot-cpp llama)
set_target_properties(cubyte_llama PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/out")
if(ANDROID)
  target_link_options(cubyte_llama PRIVATE "-Wl,-z,max-page-size=16384")
endif()
